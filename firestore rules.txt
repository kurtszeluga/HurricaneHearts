rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function userDoc() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid));
    }

    function hasUserProfile() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid));
    }

    function isActiveApprovedUser() {
      return hasUserProfile()
        && userDoc().data.active == true
        && userDoc().data.approved == true
        && userDoc().data.termsAccepted == true;
    }

    function isAdmin() {
      return isActiveApprovedUser()
        && userDoc().data.role == "admin";
    }

    function isSuperAdminEmail(email) {
      return email in [
        "hurricanehearts.admin@gmail.com",
        "kurtszeluga@gmail.com"
      ];
    }

    function isPrimaryOwnerAdmin() {
      return isAdmin()
        && isSuperAdminEmail(userDoc().data.email);
    }

    function keepsPrimaryOwnerEmail() {
      return (
        !isSuperAdminEmail(resource.data.email)
        || request.resource.data.email == resource.data.email
      )
      && (
        !isSuperAdminEmail(request.resource.data.email)
        || isSuperAdminEmail(resource.data.email)
      );
    }

    function isOwner(userId) {
      return signedIn() && request.auth.uid == userId;
    }

    match /users/{userId} {
      allow read: if isOwner(userId) || isActiveApprovedUser() || isAdmin();

      allow create: if signedIn()
        && isOwner(userId)
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.role == "resident"
        && request.resource.data.approved == false
        && request.resource.data.active == true;

      allow update: if (
          isAdmin()
          && keepsPrimaryOwnerEmail()
          && (
            isPrimaryOwnerAdmin()
            || !request.resource.data.diff(resource.data).changedKeys().hasAny([
              "role",
              "teamMember"
            ])
          )
        )
        || (
          isActiveApprovedUser()
          && isOwner(userId)
          && keepsPrimaryOwnerEmail()
          && (
            !request.resource.data.diff(resource.data).changedKeys().hasAny(["managedCategories"])
            || resource.data.teamMember == true
          )
          && !request.resource.data.diff(resource.data).changedKeys().hasAny([
            "role",
            "approved",
            "active",
            "teamMember",
            "manuallyCreated"
          ])
        )
        || (
          signedIn()
          && isOwner(userId)
          && resource.data.manuallyCreated == true
          && resource.data.active == true
          && resource.data.approved == true
          && resource.data.termsAccepted != true
          && request.resource.data.termsAccepted == true
          && request.resource.data.firstLoginTermsRequired == false
          && request.resource.data.diff(resource.data).changedKeys().hasOnly([
            "termsAccepted",
            "termsAcceptedAt",
            "termsVersion",
            "firstLoginTermsRequired"
          ])
        );

      allow delete: if isAdmin();
    }

    match /loginIds/{loginIdKey} {
      allow read: if false;

      allow create: if signedIn()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.authEmail == request.auth.token.email
        && (
          request.resource.data.email == request.auth.token.email
          || request.resource.data.email == ""
        )
        && request.resource.data.loginIdKey == loginIdKey
        && !exists(/databases/$(database)/documents/loginIds/$(loginIdKey));

      allow update: if isPrimaryOwnerAdmin();
      allow delete: if isPrimaryOwnerAdmin();
    }

    match /requests/{requestId} {
      allow read: if isActiveApprovedUser();

      allow create: if isActiveApprovedUser()
        && request.resource.data.status == "Open"
        && (
          request.resource.data.residentUid == request.auth.uid
          || isAdmin()
        );

      allow update: if isAdmin()
        || (
          isActiveApprovedUser()
          && resource.data.residentUid == request.auth.uid
          && request.resource.data.residentUid == resource.data.residentUid
        )
        || (
          isActiveApprovedUser()
          && resource.data.assignedHelperUid == request.auth.uid
          && request.resource.data.assignedHelperUid == resource.data.assignedHelperUid
        )
        || (
          isActiveApprovedUser()
          && resource.data.status == "Open"
          && request.resource.data.status == "Assigned"
          && request.resource.data.assignedHelperUid == request.auth.uid
        );

      allow delete: if isAdmin();
    }

    match /requestHistory/{historyId} {
      allow read: if isActiveApprovedUser();

      allow create: if isActiveApprovedUser()
        && request.resource.data.byUid == request.auth.uid;

      allow update, delete: if isAdmin();
    }

    match /eventHistory/{historyId} {
      allow read: if isActiveApprovedUser();
      allow create, update, delete: if isAdmin();
    }

    match /system/{docId} {
      allow read: if isActiveApprovedUser()
        && (
          (docId != "emailSettings" && docId != "signupSettings")
          || isAdmin()
        );
      allow create, update: if (
          isAdmin()
          && docId in ["activeEvent", "emailSettings"]
        )
        || (
          isPrimaryOwnerAdmin()
          && docId == "signupSettings"
        );
      allow delete: if isAdmin();
    }

    match /documents/{documentId} {
      allow read: if isActiveApprovedUser();
      allow create, update, delete: if isAdmin();
    }

    match /notifications/{notificationId} {
      allow read: if isActiveApprovedUser()
        && resource.data.toUid == request.auth.uid;

      allow create: if isActiveApprovedUser();

      allow update: if isActiveApprovedUser()
        && resource.data.toUid == request.auth.uid
        && request.resource.data.toUid == resource.data.toUid;

      allow delete: if isAdmin();
    }

    match /mailQueue/{mailId} {
      allow read: if isAdmin();

      allow create: if request.resource.data.status == "pending"
        && (
          (
            signedIn()
            && request.resource.data.accessRequestUid == request.auth.uid
            && request.resource.data.type in [
              "access-request-welcome",
              "admin-access-request"
            ]
          )
          || (
            isAdmin()
            && request.resource.data.type == "account-approved"
          )
        );

      allow update, delete: if false;
    }
  }
}
